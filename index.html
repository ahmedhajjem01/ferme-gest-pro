⚡ Configuration Gratuite en 1 Minute (Supabase) :
  1. Créez un compte gratuit sur supabase.com et créez un projet (nommez-le comme vous voulez).
  2. Dans le menu de gauche, allez dans SQL Editor, collez le script ci-dessous et cliquez sur Run.
  3. Allez dans Project Settings > API et copiez votre Project URL et anon public key ci-dessus !
create table if not exists farm_sync (
  farm_id text primary key,
  data jsonb not null,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table farm_sync enable row level security;

create policy "Public Farm Access"
  on farm_sync
  for all
  using (true)
  with check (true);
